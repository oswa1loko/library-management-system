RewriteEngine On

# Public root rewrite rules for a project deployed in /public_html/librarymanage
# Rename this file to .htaccess and place it in /public_html

# Redirect visible /librarymanage URLs to clean root paths.
RewriteCond %{THE_REQUEST} \s/+librarymanage/?([^\s?]*) [NC]
RewriteRule ^librarymanage/?(.*)$ /$1 [R=301,L]

# Serve the landing page from the app subfolder when visiting the root domain.
RewriteRule ^$ /librarymanage/index.php [L]
RewriteRule ^index\.php$ /librarymanage/index.php [L]

# Internally map clean root URLs to the deployed /librarymanage app files.
RewriteCond %{REQUEST_URI} !^/librarymanage/
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteCond %{DOCUMENT_ROOT}/librarymanage/$1 -f [OR]
RewriteCond %{DOCUMENT_ROOT}/librarymanage/$1 -d
RewriteRule ^(.*)$ /librarymanage/$1 [L]
