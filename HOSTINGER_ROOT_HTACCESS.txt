RewriteEngine On

# Public root rewrite rules for a project deployed in /public_html/librarymanage
# Rename this file to .htaccess and place it in /public_html

# Redirect direct root requests to the deployed app subfolder.
RewriteRule ^$ /librarymanage/ [R=302,L]

# Keep old direct /librarymanage URLs canonical.
RewriteCond %{THE_REQUEST} \s/+librarymanage/index\.php[\s?] [NC]
RewriteRule ^librarymanage/index\.php$ /librarymanage/ [R=301,L]
